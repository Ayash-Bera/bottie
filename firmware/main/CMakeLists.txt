if(BOTTIE_HEADLESS)
    set(app_sources "headless.cpp" "../board/simulated.cpp")
    if(BOTTIE_RECOVERED_PANEL)
        list(APPEND app_sources "../board/recovered_profile.cpp" "../board/recovered_spi.cpp" "../board/cgotchi_input.cpp")
    endif()
    if(BOTTIE_LIVE_COMPANION)
        list(APPEND app_sources "../board/live_runtime.cpp")
    endif()
else()
    set(app_sources "smoke.c")
endif()
idf_component_register(SRCS ${app_sources}
                       PRIV_REQUIRES spi_flash bottie_core esp_timer esp_driver_spi esp_driver_gpio esp_driver_i2c esp_driver_ledc esp_system heap
                       INCLUDE_DIRS "../board")
target_compile_features(${COMPONENT_LIB} PRIVATE cxx_std_17)
target_compile_options(${COMPONENT_LIB} PRIVATE -fstack-usage)
if(BOTTIE_RECOVERED_PANEL)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE BOTTIE_RECOVERED_PANEL=1)
endif()
if(BOTTIE_LIVE_COMPANION OR BOTTIE_RECOVERED_PANEL)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE BOTTIE_FRAME_WIDTH=240 BOTTIE_FRAME_HEIGHT=296)
endif()
if(BOTTIE_LIVE_COMPANION)
    target_link_libraries(${COMPONENT_LIB} PRIVATE idf::network)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE BOTTIE_LIVE_COMPANION=1)
endif()
