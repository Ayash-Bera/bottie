cmake_minimum_required(VERSION 3.22)
project(bottie_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BOTTIE_SANITIZE "Enable address and undefined behavior checks" OFF)
add_library(bottie_core ../core/parser.cpp ../core/state.cpp ../core/artwork.cpp ../core/render.cpp ../core/interaction.cpp)
target_include_directories(bottie_core PUBLIC ../core/include)
target_compile_options(bottie_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(BOTTIE_SANITIZE)
    target_compile_options(bottie_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(bottie_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(bottie_fixture fixture.cpp)
target_link_libraries(bottie_fixture PRIVATE bottie_core)
add_executable(bottie_core_tests core_tests.cpp)
target_link_libraries(bottie_core_tests PRIVATE bottie_core)
target_compile_options(bottie_core_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME bottie_core_tests COMMAND bottie_core_tests)
add_executable(bottie_panel_tests panel_tests.cpp ../board/recovered_profile.cpp)
target_include_directories(bottie_panel_tests PRIVATE ../board)
target_link_libraries(bottie_panel_tests PRIVATE bottie_core)
target_compile_options(bottie_panel_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME bottie_panel_tests COMMAND bottie_panel_tests)
add_executable(bottie_cache_tests cache_tests.cpp)
target_include_directories(bottie_cache_tests PRIVATE ../board)
target_link_libraries(bottie_cache_tests PRIVATE bottie_core)
target_compile_options(bottie_cache_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME bottie_cache_tests COMMAND bottie_cache_tests)
add_executable(bottie_band_tests band_tests.cpp)
target_link_libraries(bottie_band_tests PRIVATE bottie_core)
target_compile_options(bottie_band_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME bottie_band_tests COMMAND bottie_band_tests)
add_executable(bottie_input_tests input_tests.cpp)
target_include_directories(bottie_input_tests PRIVATE ../board)
target_link_libraries(bottie_input_tests PRIVATE bottie_core)
target_compile_options(bottie_input_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME bottie_input_tests COMMAND bottie_input_tests)
